-- =========================================================
-- 1.사용자와 권한 조회
-- 사용자별 권한 확인
-- =========================================================
SELECT
    u.user_id,
    u.user_name,
    ur.role_name
FROM `USER` u
JOIN USER_ROLE ur
    ON u.role_id = ur.role_id
ORDER BY u.user_id;

-- =========================================================
-- 2.방별 설치된 센서 조회
-- 각 방에 어떤 센서가 설치되어 있는지 확인
-- =========================================================
SELECT
    r.room_name,
    s.sensor_name,
    st.type_name,
    st.unit
FROM SENSOR s
JOIN ROOM r
    ON s.room_id = r.room_id
JOIN SENSOR_TYPE st
    ON s.sensor_type_id = st.sensor_type_id
ORDER BY r.room_id, s.sensor_id;

-- =========================================================
-- 3.방별 설치된 액추에이터 조회
-- 각 방에 어떤 제어 장치가 설치되어 있는지 확인
-- =========================================================
SELECT
    r.room_name,
    a.actuator_name,
    at.type_name
FROM ACTUATOR a
JOIN ROOM r
    ON a.room_id = r.room_id
JOIN ACTUATOR_TYPE at
    ON a.actuator_type_id = at.actuator_type_id
ORDER BY r.room_id, a.actuator_id;

-- =========================================================
-- 4.전체 센서 측정 기록 조회
-- 센서값이 시간에 따라 기록되는지 확인
-- =========================================================
SELECT
    r.room_name,
    s.sensor_name,
    st.type_name,
    sr.measured_value,
    st.unit,
    sr.measured_time
FROM SENSOR_READING sr
JOIN SENSOR s
    ON sr.sensor_id = s.sensor_id
JOIN SENSOR_TYPE st
    ON s.sensor_type_id = st.sensor_type_id
JOIN ROOM r
    ON s.room_id = r.room_id
ORDER BY sr.measured_time, r.room_id, s.sensor_id;

-- =========================================================
-- 5.센서별 최신 측정값 조회
-- 현재 센서값 조회
-- =========================================================
SELECT
    r.room_name,
    s.sensor_name,
    st.type_name,
    sr.measured_value,
    st.unit,
    sr.measured_time
FROM SENSOR_READING sr
JOIN SENSOR s
    ON sr.sensor_id = s.sensor_id
JOIN SENSOR_TYPE st
    ON s.sensor_type_id = st.sensor_type_id
JOIN ROOM r
    ON s.room_id = r.room_id
WHERE sr.measured_time = (
    SELECT MAX(sr2.measured_time)
    FROM SENSOR_READING sr2
    WHERE sr2.sensor_id = sr.sensor_id
)
ORDER BY r.room_id, s.sensor_id;

-- =========================================================
-- 6.특정 방의 최신 센서값 조회
-- 특정 방의 현재 환경 상태 확인
-- =========================================================
SELECT
    r.room_name,
    s.sensor_name,
    st.type_name,
    sr.measured_value,
    st.unit,
    sr.measured_time
FROM SENSOR_READING sr
JOIN SENSOR s
    ON sr.sensor_id = s.sensor_id
JOIN SENSOR_TYPE st
    ON s.sensor_type_id = st.sensor_type_id
JOIN ROOM r
    ON s.room_id = r.room_id
WHERE r.room_name = '거실'
  AND sr.measured_time = (
    SELECT MAX(sr2.measured_time)
    FROM SENSOR_READING sr2
    WHERE sr2.sensor_id = sr.sensor_id
)
ORDER BY s.sensor_id;

-- =========================================================
-- 7.방별 평균 온도 조회
-- 센서 데이터를 활용한 간단한 분석
-- =========================================================
SELECT
    r.room_name,
    ROUND(AVG(sr.measured_value), 2) AS avg_temperature
FROM SENSOR_READING sr
JOIN SENSOR s
    ON sr.sensor_id = s.sensor_id
JOIN SENSOR_TYPE st
    ON s.sensor_type_id = st.sensor_type_id
JOIN ROOM r
    ON s.room_id = r.room_id
WHERE st.type_name = 'temperature'
GROUP BY r.room_id, r.room_name
ORDER BY r.room_id;

-- =========================================================
-- 8.액추에이터별 최신 상태 조회
-- 현재 장치 상태 확인
-- =========================================================
SELECT
    r.room_name,
    a.actuator_name,
    at.type_name,
    asl.state_value,
    asl.changed_time
FROM ACTUATOR_STATE_LOG asl
JOIN ACTUATOR a
    ON asl.actuator_id = a.actuator_id
JOIN ACTUATOR_TYPE at
    ON a.actuator_type_id = at.actuator_type_id
JOIN ROOM r
    ON a.room_id = r.room_id
WHERE asl.changed_time = (
    SELECT MAX(asl2.changed_time)
    FROM ACTUATOR_STATE_LOG asl2
    WHERE asl2.actuator_id = asl.actuator_id
)
ORDER BY r.room_id, a.actuator_id;

-- =========================================================
-- 9.현재 작동 중인 장치 조회
-- 현재 켜져 있거나 작동 중인 장치만 확인
-- =========================================================
SELECT
    r.room_name,
    a.actuator_name,
    at.type_name,
    asl.state_value,
    asl.changed_time
FROM ACTUATOR_STATE_LOG asl
JOIN ACTUATOR a
    ON asl.actuator_id = a.actuator_id
JOIN ACTUATOR_TYPE at
    ON a.actuator_type_id = at.actuator_type_id
JOIN ROOM r
    ON a.room_id = r.room_id
WHERE asl.changed_time = (
    SELECT MAX(asl2.changed_time)
    FROM ACTUATOR_STATE_LOG asl2
    WHERE asl2.actuator_id = asl.actuator_id
)
  AND asl.state_value IN ('ON', 'COOLING', 'HEATING', 'OPEN')
ORDER BY r.room_id, a.actuator_id;

-- =========================================================
-- 10.방별 현재 상태 조회
-- 방이 비었는지, 사용중인지 등을 확인
-- =========================================================
SELECT
    r.room_name,
    rst.status_name,
    rsl.changed_time
FROM ROOM_STATUS_LOG rsl
JOIN ROOM r
    ON rsl.room_id = r.room_id
JOIN ROOM_STATUS_TYPE rst
    ON rsl.status_id = rst.status_id
WHERE rsl.changed_time = (
    SELECT MAX(rsl2.changed_time)
    FROM ROOM_STATUS_LOG rsl2
    WHERE rsl2.room_id = rsl.room_id
)
ORDER BY r.room_id;

-- =========================================================
-- 11.사용자 현재 위치 조회
-- 사용자별 현재 위치 확인
-- =========================================================
SELECT
    u.user_name,
    COALESCE(r.room_name, '위치 없음') AS current_location
FROM USER_LOCATION ul
JOIN `USER` u
    ON ul.user_id = u.user_id
LEFT JOIN ROOM r
    ON ul.room_id = r.room_id
ORDER BY u.user_id;

-- =========================================================
-- 12.방별 현재 위치한 사용자 조회
-- 방마다 현재 누가 있는지 확인
-- =========================================================
SELECT
    r.room_name,
    COALESCE(GROUP_CONCAT(u.user_name ORDER BY u.user_name SEPARATOR ', '), '없음') AS users_in_room
FROM ROOM r
LEFT JOIN USER_LOCATION ul
    ON r.room_id = ul.room_id
LEFT JOIN `USER` u
    ON ul.user_id = u.user_id
GROUP BY r.room_id, r.room_name
ORDER BY r.room_id;

-- =========================================================
-- 13.예약 전체 조회
-- 전체 방 예약 현황 확인
-- =========================================================
SELECT
    rr.reservation_id,
    u.user_name,
    r.room_name,
    rr.start_time,
    rr.end_time,
    rr.purpose,
    rr.reservation_status
FROM ROOM_RESERVATION rr
JOIN `USER` u
    ON rr.user_id = u.user_id
JOIN ROOM r
    ON rr.room_id = r.room_id
ORDER BY rr.start_time;

-- =========================================================
-- 14.특정 시점에 예약 중인 방 조회
-- 특정 시간 기준으로 어떤 방이 예약 중인지 확인
-- =========================================================
SELECT
    r.room_name,
    u.user_name,
    rr.start_time,
    rr.end_time,
    rr.purpose,
    rr.reservation_status
FROM ROOM_RESERVATION rr
JOIN ROOM r
    ON rr.room_id = r.room_id
JOIN `USER` u
    ON rr.user_id = u.user_id
WHERE '2026-05-04 14:30:00' BETWEEN rr.start_time AND rr.end_time
  AND rr.reservation_status = 'RESERVED'
ORDER BY rr.start_time;

-- =========================================================
-- 15.예약 가능한 방 조회
-- 원하는 시간대에 예약 가능한 방 찾기
-- =========================================================
SELECT
    r.room_id,
    r.room_name
FROM ROOM r
WHERE NOT EXISTS (
    SELECT 1
    FROM ROOM_RESERVATION rr
    WHERE rr.room_id = r.room_id
      AND rr.reservation_status = 'RESERVED'
      AND rr.start_time < '2026-05-04 16:00:00'
      AND rr.end_time > '2026-05-04 14:00:00'
)
ORDER BY r.room_id;
